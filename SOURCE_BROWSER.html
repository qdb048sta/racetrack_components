47def load_track(path: str | Path) -> tuple[FloatArray, FloatArray, float]:
48 """Return unclosed rows, chord-distance labels u, and closed polyline length U."""
49 a = np.loadtxt(path, delimiter=',', comments='#', ndmin=2)
50 if a.shape[1] < 4 or a.shape[0] < 8 or not np.isfinite(a).all():
51 raise ValueError('Need at least eight finite rows: x,y,right_width,left_width.')
52 a = a[:, :4].astype(float)
53 if np.linalg.norm(a[0,:2] - a[-1,:2]) < 1e-8:
54 if not np.allclose(a[0,2:], a[-1,2:], atol=1e-6):
55 raise ValueError('Repeated endpoint has inconsistent widths.')
56 a = a[:-1]
57 if np.any(a[:,2:] <= 0):
58 raise ValueError('All half-widths must be positive.')
59 d = np.linalg.norm(np.roll(a[:,:2], -1, axis=0) - a[:,:2], axis=1)
60 if np.any(d < 1e-8):
61 raise ValueError('Remove consecutive duplicate coordinates before fitting.')
62 if d[-1] > 5*np.median(d[:-1]):
63 raise ValueError('Closing gap is unexpectedly large: check ordering/closure.')
64 return a, np.r_[0.0,np.cumsum(d[:-1])], float(d.sum())

74 def __call__(self, u: Any, derivative: int=0) -> FloatArray:
75 u = np.asarray(u, dtype=float)

82 def curvature(self, u: Any) -> FloatArray:
83 if self.method == 'linear':
84 return np.full(np.asarray(u).shape, np.nan)
85 a,b = self(u,1), self(u,2)
86 speed = np.linalg.norm(a,axis=-1)
87 cross = a[...,0]*b[...,1]-a[...,1]*b[...,0]
88 return np.divide(cross, speed**3, out=np.full_like(cross,np.nan), where=speed>1e-10)

91def smoothing_factor(x: FloatArray, period: float, lam: float) -> tuple[Any, FloatArray]:
92 """Exact periodic cubic bending penalty, in a nodal/cardinal spline basis.
93
94 Minimize sum_i ||v_i-y_i||^2 + lam * integral ||C_v''(u)||^2 du.
95 C_v is the periodic cubic interpolant through unknown fitted values v_i.
96 Two-point Gauss integration is exact for products of linear second derivatives.
97 The parameter u is chord-distance in metres, so lam has units m^3 here.
98 This is not FITPACK's residual-budget s and is not a curvature-energy penalty.
99 """
100 if lam < 0:
101 raise ValueError('Smoothing lambda must be nonnegative.')
102 n=len(x); closed_x=np.r_[x,x[0]+period]
103 eye=np.eye(n)
104 cardinal=CubicSpline(closed_x,np.vstack([eye,eye[0]]),bc_type='periodic',axis=0)
105 h=np.diff(closed_x); mid=(closed_x[:-1]+closed_x[1:])/2
106 nodes=(mid[:,None]+h[:,None]/2*np.array([-1,1])[None,:]/np.sqrt(3)).ravel()
107 weights=np.repeat(h/2,2)
108 D=cardinal(nodes,2)
109 penalty=D.T @ (weights[:,None]*D)
110 penalty=(penalty+penalty.T)/2
111 return cho_factor(np.eye(n)+lam*penalty, lower=True, check_finite=False), closed_x

114def make_curve(method: str, x: FloatArray, y: FloatArray, period: float,
115 lam: float=100.0, factor: Any=None) -> Curve:
116 """x contains unique points in increasing cyclic order, without repeated endpoint."""
117 if method not in METHODS:
118 raise ValueError(f'Unknown interpolation method: {method}')
119 xc=np.r_[x,x[0]+period]; yc=np.vstack([y,y[0]])
120 if method == 'linear':
121 f=make_interp_spline(xc,yc,k=1,axis=0)
122 elif method == 'periodic_cubic':
123 f=CubicSpline(xc,yc,bc_type='periodic',axis=0)
124 elif method == 'periodic_quintic':
125 f=make_interp_spline(xc,yc,k=5,bc_type='periodic',axis=0)
126 elif method == 'periodic_smoothing_cubic':
127 if factor is None: factor=smoothing_factor(x,period,lam)[0]
128 fitted=cho_solve(factor,y,check_finite=False)
129 f=CubicSpline(xc,np.vstack([fitted,fitted[0]]),bc_type='periodic',axis=0)
130 else:
131 # Cyclic neighbours are needed to compute the END slopes. Merely wrapping
132 # evaluation points with modulo does not make ordinary endpoint slopes match.
133 k=3
134 xp=np.r_[x[-k:]-period,x,x[:k]+period]
135 yp=np.vstack([y[-k:],y,y[:k]])
136 if method == 'pchip_cyclic':
137 f=PchipInterpolator(xp,yp,axis=0)
138 else:
139 f=Akima1DInterpolator(xp,yp,axis=0,
140 method='makima' if method=='makima_cyclic' else 'akima')
141 return Curve(f,float(x[0]),float(period),method)

148def seam_and_knots(curve: Curve, x: FloatArray) -> dict[str,float]:
149 h=np.diff(np.r_[x,x[0]+curve.period])
150 eps=min(h)*1e-7
151 am,ap=curve(x-eps,1),curve(x+eps,1)
152 jumps=np.abs(wrap_angle(np.arctan2(am[:,1],am[:,0])-np.arctan2(ap[:,1],ap[:,0])))
153 out={'max_heading_jump_rad':float(jumps.max()),
154 'seam_heading_jump_rad':float(jumps[0])}
155 if curve.method != 'linear':
156 kjump=np.abs(curve.curvature(x-eps)-curve.curvature(x+eps))
157 out.update(max_curvature_jump_1pm=float(np.nanmax(kjump)),
158 seam_curvature_jump_1pm=float(kjump[0]))
159 else:
160 out.update(max_curvature_jump_1pm=float('nan'),seam_curvature_jump_1pm=float('nan'))
161 return out

180class PeriodicTrack:
181 """Road geometry API for replacing an open graph road y=c(X).
182
183 Uses all source points, an exact periodic cubic (or quintic), and cyclic PCHIP
184 half-widths. Friction multiplier=1, height=0 are EXPLICIT demonstration defaults,
185 not measurements. Exact interpolation still doesn't validate physical borders.
186 General centreline smoothing is deliberately disallowed here because changing
187 the reference curve requires preserving/reprojecting the original boundaries.
188 """
189 def __init__(self, path: str | Path, method: str='periodic_cubic'):
190 if method not in ('periodic_cubic','periodic_quintic'):
191 raise ValueError('Road API only uses exact periodic cubic/quintic; see boundary warning.')
192 self.data,self.u,self.U=load_track(path)
193 self.curve=make_curve(method,self.u,self.data[:,:2],self.U)
194 self.width=make_curve('pchip_cyclic',self.u,self.data[:,2:],self.U)
195 xc=np.r_[self.u,self.U]
196 # Eight subintervals per original segment, five-point Gauss quadrature each.
197 uq=np.concatenate([a+(b-a)*np.arange(8)/8 for a,b in zip(xc[:-1],xc[1:])]+[np.array([self.U])])
198 g,w=np.polynomial.legendre.leggauss(5)
199 h=np.diff(uq);mid=(uq[:-1]+uq[1:])/2
200 nodes=(mid[:,None]+h[:,None]/2*g).ravel()
201 speed=np.linalg.norm(self.curve(nodes,1),axis=1).reshape(-1,5)
202 seg=h/2*(speed@w)
203 self.s_grid=np.r_[0,np.cumsum(seg)];self.u_grid=uq
204 self.length=float(self.s_grid[-1])
205 self.u_from_s=PchipInterpolator(self.s_grid,uq)
206 self.s_from_u=PchipInterpolator(uq,self.s_grid)
207
208 def sample(self,s: Any) -> dict[str,FloatArray]:
209 s=np.asarray(s,dtype=float);smod=np.mod(s,self.length)
210 u=self.u_from_s(smod)
211 c=self.curve(u);d=self.curve(u,1);speed=np.linalg.norm(d,axis=-1)
212 if np.any(speed<1e-8): raise RuntimeError('Degenerate tangent in interpolated track.')
213 t=d/speed[...,None];n=np.stack([-t[...,1],t[...,0]],axis=-1)
214 widths=self.width(u);wr,wl=widths[...,0],widths[...,1]
215 return {'s_m':smod,'u_chord_m':u,'center':c,'tangent':t,'normal_left':n,
216 'heading_rad':np.arctan2(t[...,1],t[...,0]),
217 'curvature_1pm':self.curve.curvature(u),'width_right_m':wr,'width_left_m':wl,
218 'boundary_right':c-wr[...,None]*n,'boundary_left':c+wl[...,None]*n,
219 'friction_multiplier_assumed':np.ones_like(s),'height_m_assumed':np.zeros_like(s)}

221 def project(self,xy: Any,hint_s: float|None=None,window_m: float=60.0) -> dict[str,float]:
222 """Numerical closest-point projection; hint reduces branch switching.
223
224 Returns station unwrapped close to hint_s when provided. This is a local
225 numerical projection, not a theorem of globally unique Frenet coordinates.
226 """
227 point=np.asarray(xy,dtype=float)
228 if point.shape!=(2,) or not np.isfinite(point).all(): raise ValueError('xy must be two finite numbers.')
229 if hint_s is None:
230 sg=np.linspace(0,self.length,math.ceil(self.length),endpoint=False)
231 else:
232 sg=np.linspace(hint_s-window_m,hint_s+window_m,max(50,math.ceil(2*window_m)))
233 c=self.sample(sg)['center'];i=int(np.argmin(np.sum((c-point)**2,axis=1)))
234 step=float(sg[1]-sg[0]);g0=float(sg[i])
235 res=minimize_scalar(lambda s:float(np.sum((self.sample(s)['center']-point)**2)),
236 bounds=(g0-2*step,g0+2*step),method='bounded',options={'xatol':1e-9})
237 q=self.sample(res.x);e=float(np.dot(point-q['center'],q['normal_left']))
238 station=float(res.x if hint_s is not None else res.x%self.length)
239 return {'s_m':station,'e_left_m':e,'distance_m':float(np.sqrt(res.fun))}

269 'max_no_slip_steering_for_3p2m_wheelbase_rad':float(np.arctan(3.2*peak)),
270 'no_slip_min_radius_with_3p2m_wheelbase_and_0p30rad_steering_m':float(3.2/np.tan(.30)),
271 'min_left_offset_jacobian_1_minus_kappa_wleft':float(np.min(1-k*wl)),
272 'min_right_offset_jacobian_1_plus_kappa_wright':float(np.min(1+k*wr))}

48def smoothing_factor(x: FloatArray, period: float, lam: float) -> tuple[Any, FloatArray]:
49 """Exact periodic cubic bending penalty, in a nodal/cardinal spline basis.
50
51 Minimize sum_i ||v_i-y_i||^2 + lam * integral ||C_v''(u)||^2 du.
52 C_v is the periodic cubic interpolant through unknown fitted values v_i.
53 Two-point Gauss integration is exact for products of linear second derivatives.
54 The parameter u is chord-distance in metres, so lam has units m^3 here.
55 This is not FITPACK's residual-budget s and is not a curvature-energy penalty.
56 """
57 if lam < 0:
58 raise ValueError('Smoothing lambda must be nonnegative.')
59 n=len(x); closed_x=np.r_[x,x[0]+period]
60 eye=np.eye(n)
61 cardinal=CubicSpline(closed_x,np.vstack([eye,eye[0]]),bc_type='periodic',axis=0)
62 h=np.diff(closed_x); mid=(closed_x[:-1]+closed_x[1:])/2
63 nodes=(mid[:,None]+h[:,None]/2*np.array([-1,1])[None,:]/np.sqrt(3)).ravel()
64 weights=np.repeat(h/2,2)
65 D=cardinal(nodes,2)
66 penalty=D.T @ (weights[:,None]*D)
67 penalty=(penalty+penalty.T)/2
68 return cho_factor(np.eye(n)+lam*penalty, lower=True, check_finite=False), closed_x

200class VehicleModel:
201 """Modified source equations with a differentiable CasADi representation.
202
203 Spatial state z=[Y,psi,vx,vy,r,delta] (example1), plus [ax_load,ay_load]
204 (example2). Time-domain Xdot is evaluated separately; spatial derivative
205 is dz/dX=(dz/dt)/Xdot. Input u=[steering_rate, common_wheel_slip_ratio].
206 Example2 is a PLANAR reduction of double-track, NOT its 28-state plant.
207 """
208 def __init__(self, p: CarParams, road: Road):
209 self.p, self.road = p, road
210 self.nx = 6 if road.example == 1 else 8
211 self.names = ["Y_m", "psi_rad", "vx_mps", "vy_mps", "r_radps", "delta_rad"]
212 self.scale = np.array([5.0, 0.3, 20.0, 2.0, 0.5, 0.3])
213 if self.nx == 8:
214 self.names += ["load_ax_mps2", "load_ay_mps2"]
215 self.scale = np.r_[self.scale, 4.0, 4.0]
216 self.u_scale = np.array([p.steer_rate_max, 0.05])
217 self.alpha_max = 0.065 if road.example == 1 else 0.16
218 self.kappa_max = 0.07 if road.example == 1 else 0.12
219 self.util_max = 0.95 if road.example == 1 else 0.985
220 self._build()
221
222 def _tire(self, kappa: Any, alpha: Any, fz: Any, mu: Any) -> tuple[Any, Any, Any]:
223 p = self.p
224 if self.road.example == 1:
225 return mu*fz*p.c_kappa*kappa, mu*fz*p.c_alpha*alpha, mu*fz
226 # Adapted from mf_simple_extended.cpp; this regularization is a CHANGE.
227 lat = ca.tan(alpha)
228 slip = ca.sqrt(kappa*kappa + lat*lat + p.slip_epsilon**2)
229 nominal = p.mass*p.gravity/4.0
230 d = p.d0 + p.pdy2*(fz - nominal)/nominal
231 def mf(b: float, c: float, e: float) -> Any:
232 bs = b*slip
233 return ca.sin(c*ca.atan(bs-e*(bs-ca.atan(bs))))
234 peak = mu*fz*d
235 return peak*mf(p.bx,p.cx,p.ex)*kappa/slip, peak*mf(p.by,p.cy,p.ey)*lat/slip, peak

237 def _build(self) -> None:
238 p = self.p
239 x = ca.SX.sym("X")
240 z = ca.SX.sym("z", self.nx)
241 u = ca.SX.sym("u", 2)
242 y, psi, vx, vy, r, delta = [z[j] for j in range(6)]
243 nu, kappa = u[0], u[1]
244 front = p.mass*p.gravity*p.lr/p.wheelbase
245 rear = p.mass*p.gravity*p.lf/p.wheelbase
246 b = 0.0 if self.road.example == 1 else p.track
247 fzs = [front/2, front/2, rear/2, rear/2]
248 if self.nx == 8:
249 # NEW reduction of roll/pitch/suspension, not upstream exact physics.
250 ax_load, ay_load = z[6], z[7]
251 front = front - p.mass*p.cg_height*ax_load/p.wheelbase
252 rear = rear + p.mass*p.cg_height*ax_load/p.wheelbase
253 transfer_f = p.mass*p.cg_height*(p.lr/p.wheelbase)*ay_load/p.track
254 transfer_r = p.mass*p.cg_height*(p.lf/p.wheelbase)*ay_load/p.track
255 fzs = [front/2-transfer_f, front/2+transfer_f,
256 rear/2-transfer_r, rear/2+transfer_r]
257 fxsum, fysum, moment = 0, 0, 0
258 alphas, forces_x, forces_y, peaks, mus, tire_u = [], [], [], [], [], []
259 for i, (xi, yi) in enumerate([(p.lf,b/2),(p.lf,-b/2),(-p.lr,b/2),(-p.lr,-b/2)]):
260 di = delta if i < 2 else 0.0
261 ub, vb = vx-r*yi, vy+r*xi
262 ut = ca.cos(di)*ub+ca.sin(di)*vb
263 vt = -ca.sin(di)*ub+ca.cos(di)*vb
264 # Exact upstream branch for forward speeds; feasibility requires ut>=3.
265 alpha = -ca.atan2(vt, ca.fmax(1.0, ut))
266 wheel_x = x + ca.cos(psi)*xi-ca.sin(psi)*yi
267 wheel_y = y + ca.sin(psi)*xi+ca.cos(psi)*yi
268 mu = self.road.friction(wheel_x,wheel_y,ca)
269 fx, fy, peak = self._tire(kappa,alpha,fzs[i],mu)
270 fx_b = ca.cos(di)*fx-ca.sin(di)*fy
271 fy_b = ca.sin(di)*fx+ca.cos(di)*fy
272 fxsum += fx_b
273 fysum += fy_b
274 moment += xi*fy_b-yi*fx_b
275 alphas.append(alpha); forces_x.append(fx); forces_y.append(fy)
276 peaks.append(peak); mus.append(mu); tire_u.append(ut)
277 # No downforce/wind; explicitly simplified aero. Rolling loss moved from
278 # native wheel-load torque into the body equation after removing wheels.
279 drag = 0.5*p.rho_air*p.cd_area*vx*ca.sqrt(vx*vx+1e-8)
280 resistance = p.rolling_coefficient*p.mass*p.gravity
281 total_fx = fxsum-drag-resistance
282 total_fy = fysum
283 ax, ay = total_fx/p.mass, total_fy/p.mass
284 progress = vx*ca.cos(psi)-vy*ca.sin(psi)
285 ydot = vx*ca.sin(psi)+vy*ca.cos(psi)
286 timed = ca.vertcat(ydot,r,ax+r*vy,ay-r*vx,moment/p.iz,nu)
287 if self.nx == 8:
288 timed = ca.vertcat(timed,(ax-z[6])/p.load_tau,(ay-z[7])/p.load_tau)
289 # Numerical extension outside feasible domain. progress>=4 is enforced.
290 denom = ca.fmax(1.0,progress)
291 self.time_rhs = ca.Function("time_rhs",[x,z,u],[timed,progress])
292 self.spatial = ca.Function("spatial",[x,z,u],[timed/denom,1/denom])
293 alpha_vec = ca.vertcat(*alphas)
294 fxv, fyv, pk = ca.vertcat(*forces_x), ca.vertcat(*forces_y), ca.vertcat(*peaks)
295 util2 = (fxv**2+fyv**2)/(pk**2+1e-10)
296 room = self.road.half_width-p.enclosing_radius*math.sqrt(1+self.road.slope_bound**2)
297 error = y-self.road.center(x,ca)
298 beta = ca.atan2(vy, vx)
299 nominal_power_proxy = sum(forces_x)*vx # deliberate reduced-powertrain proxy
300 cons = ca.vertcat(
301 (room-p.safety_margin-error)/self.road.half_width,
302 (room-p.safety_margin+error)/self.road.half_width,
303 (vx-p.speed_min)/20, (p.speed_max-vx)/20,
304 (progress-4.0)/20,
305 (p.beta_max-beta), (p.beta_max+beta),
306 (0.60-psi), (0.60+psi),
307 (0.90-r), (0.90+r),
308 (p.steer_max-delta), (p.steer_max+delta),
309 (p.acceleration_max-ax)/10, (p.braking_max+ax)/10,
310 (p.power_max-nominal_power_proxy)/p.power_max,
311 self.alpha_max-alpha_vec,self.alpha_max+alpha_vec,
312 self.util_max**2-util2,
313 (ca.vertcat(*fzs)-p.min_wheel_load)/2000,
314 (ca.vertcat(*tire_u)-3.0)/20,
315 )
316 self.constraints = ca.Function("constraints",[x,z,u],[cons])
317 self.diagnostics = ca.Function("diagnostics",[x,z,u],[
318 ca.vertcat(progress,ax,ay,beta,error,room-ca.fabs(error),
319 alpha_vec,fxv,fyv,ca.vertcat(*fzs),ca.vertcat(*mus),ca.sqrt(util2),nominal_power_proxy)])
320 self.tire_test = ca.Function("tire_test",[ca.SX.sym("dummy")],[0])
321 kk, aa, zz, mm = ca.SX.sym("kap"),ca.SX.sym("alpha"),ca.SX.sym("fz"),ca.SX.sym("mu")
322 tx,ty,tp = self._tire(kk,aa,zz,mm)
323 self.tire = ca.Function("tire",[kk,aa,zz,mm],[ca.vertcat(tx,ty,tp)])
324 self.jacobian = ca.Function("jacobian",[x,z,u],[ca.jacobian(timed,z)])

326 def initial_state(self, speed: float = 8.0) -> np.ndarray:
327 z = np.zeros(self.nx)
328 z[0] = self.road.center(0.0)
329 z[1] = math.atan(self.road.slope(0.0))
330 z[2] = speed
331 return z

333 def make_step(self, ds: float, subdivisions: int) -> ca.Function:
334 """RK4 multiple-shooting transition, with integrated time and guards."""
335 x = ca.MX.sym("X")
336 z = ca.MX.sym("z",self.nx)
337 u = ca.MX.sym("u",2)
338 zz = z
339 dt = 0
340 checks = [self.constraints(x,zz,u)-self.p.constraint_backoff]
341 h = ds/subdivisions
342 for j in range(subdivisions):
343 xx = x+j*h
344 k1,t1 = self.spatial(xx,zz,u)
345 k2,t2 = self.spatial(xx+h/2,zz+h*k1/2,u)
346 k3,t3 = self.spatial(xx+h/2,zz+h*k2/2,u)
347 k4,t4 = self.spatial(xx+h,zz+h*k3,u)
348 zz = zz+h*(k1+2*k2+2*k3+k4)/6
349 dt += h*(t1+2*t2+2*t3+t4)/6
350 checks.append(self.constraints(xx+h,zz,u)-self.p.constraint_backoff)
351 return ca.Function("rk4_step",[x,z,u],[zz,dt,ca.vertcat(*checks)])

354class MinimumTimeMPC:
355 """Shrinking, full-remaining-course, spatial MPC; local NLP via IPOPT."""
356 def __init__(self, model: VehicleModel, cells: int, subdivisions: int):
357 self.model = model
358 self.cells = cells
359 self.ds = model.road.length/cells
360 self.step = model.make_step(self.ds,subdivisions)
361 self.previous: tuple[np.ndarray,np.ndarray] | None = None

363 def guess(self, current_x: float, n: int, actual: np.ndarray) -> tuple[np.ndarray,np.ndarray]:
364 m = self.model
365 if self.previous is not None:
366 oldz,oldu = self.previous
367 z = oldz[:,1:].copy()
368 u = oldu[:,1:].copy()
369 if z.shape == (m.nx,n+1) and u.shape == (2,n):
370 z[:,0] = actual
371 return z,u
372 xgrid = current_x+self.ds*np.arange(n+1)
373 speed = max(8.0,actual[2])
374 z = np.zeros((m.nx,n+1))
375 z[0] = m.road.center(xgrid)
376 heading = np.arctan(m.road.slope(xgrid))
377 curvature = np.gradient(heading,self.ds)
378 z[2] = speed
379 z[4] = speed*curvature
380 alpha_rear = speed**2*curvature/(m.p.c_alpha*m.p.gravity)
381 z[3] = m.p.lr*z[4]-speed*alpha_rear
382 z[1] = heading-np.arctan2(z[3],z[2])
383 z[5] = m.p.wheelbase*curvature
384 if m.nx == 8:
385 z[7] = speed**2*curvature
386 z[:,0] = actual
387 u = np.zeros((2,n))
388 u[0] = np.clip(np.diff(z[5])*speed/self.ds,-.3,.3)
389 u[1] = 0.002
390 return z,u

392 def solve(self,current_x: float,n: int,actual: np.ndarray) -> tuple[np.ndarray,dict[str,Any]]:
393 m = self.model
394 begin = time.perf_counter()
395 zs = ca.MX.sym("zs",m.nx,n+1)
396 us = ca.MX.sym("us",2,n)
397 z0 = ca.MX.sym("z0",m.nx)
398 z = ca.repmat(ca.DM(m.scale),1,n+1)*zs
399 u = ca.repmat(ca.DM(m.u_scale),1,n)*us
400 eq = [(z[:,0]-z0)/ca.DM(m.scale)]
401 ineq = []
402 objective = 0
403 for k in range(n):
404 nxt,dt,ck = self.step(current_x+k*self.ds,z[:,k],u[:,k])
405 eq.append((z[:,k+1]-nxt)/ca.DM(m.scale))
406 ineq.append(ck)
407 objective += dt
408 gg = ca.vertcat(*eq,*ineq)
409 neq = m.nx*(n+1)
410 ng = int(gg.numel())
411 w = ca.vertcat(ca.vec(zs),ca.vec(us))
412 nlp = {"x":w,"p":z0,"f":objective,"g":gg}
413 options = {"print_time":False,"ipopt.print_level":0,"ipopt.sb":"yes",
414 "ipopt.max_iter":600,"ipopt.tol":2e-8,"ipopt.acceptable_tol":2e-7,
415 "ipopt.bound_relax_factor":0.0}
416 solver = ca.nlpsol("minimum_time", "ipopt", nlp, options)
417 lo_z = np.array([-20,-.60,m.p.speed_min,-5,-.90,-m.p.steer_max]+([-8,-10] if m.nx==8 else []))/m.scale
418 hi_z = np.array([20,.60,m.p.speed_max,5,.90,m.p.steer_max]+([8,10] if m.nx==8 else []))/m.scale
419 lbx = np.r_[np.tile(lo_z,n+1),np.tile([-1,-m.kappa_max/m.u_scale[1]],n)]
420 ubx = np.r_[np.tile(hi_z,n+1),np.tile([1,m.kappa_max/m.u_scale[1]],n)]
421 zg,ug = self.guess(current_x,n,actual)
422 initial = np.r_[(zg/m.scale[:,None]).ravel(order="F"),(ug/m.u_scale[:,None]).ravel(order="F")]
423 lbg=np.zeros(ng);ubg=np.r_[np.zeros(neq),np.full(ng-neq,np.inf)]
424 result = solver(x0=initial,p=actual,lbx=lbx,ubx=ubx,lbg=lbg,ubg=ubg)
425 stats = solver.stats()
426 sol = np.asarray(result["x"]).reshape(-1)
427 gv = np.asarray(result["g"]).reshape(-1)
428 violation = max(float(np.max(np.abs(gv[:neq]))),float(max(0,-np.min(gv[neq:]))),
429 float(max(0,np.max(lbx-sol))),float(max(0,np.max(sol-ubx))))
430 if not stats["success"] or violation > 2e-5:
431 raise RuntimeError(f"NLP failed at X={current_x:.3f}: {stats['return_status']}; violation={violation:.3g}. No fallback control was applied.")
432 cut = m.nx*(n+1)
433 zz = sol[:cut].reshape((m.nx,n+1),order="F")*m.scale[:,None]
434 uu = sol[cut:].reshape((2,n),order="F")*m.u_scale[:,None]
435 self.previous=(zz,uu)
436 return uu[:,0], {"X_m":current_x,"horizon_intervals":n,"objective_remaining_time_s":float(result["f"]),
437 "iterations":int(stats["iter_count"]),"return_status":stats["return_status"],
438 "max_scaled_constraint_violation":violation,
439 "build_and_solve_wall_s":time.perf_counter()-begin}
